SELECT DISTINCT p1.email FROM Person p1 JOIN Person p2 ON p1.email = p2.email AND p1.id <> p2.id;

--Approach-4 CTC Best case

with prep as (
    select 
        id,
        email,
        row_number() over(partition by email order by id )as rn
    From 
        Person 
)
select email from prep where rn = 2

-- Approach - 5

Select distinct email as Email from (Select *, dense_rank() over(partition by email order by id) as drnk from Person) x
