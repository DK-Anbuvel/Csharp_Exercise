6. Find highest salary in each department

  
select departmentid,emppname,salary from (
  select departmentid,empname,salary, DENSE_RANK over (
   partition by departmentid
   order by salary desc
   ) as Rankno
   from employees
) E where RankNo =1;
 
 partition by -> it creates a separate ranking for each department

7. Find top 3 salaries in each department

   ) as Rankno

 14 Difference between ROW_NUMBER, RANK and DENSE_RANK

 example
 10000
 10000
 200
 300
 ROW_NUMBER() OVER (ORDER BY Salary DESC) --> Every row gets a unique number.
  resul:-
  1000 -1
  1000 -2
  200  -3
  300  -4
 RANK() ---> There is a gap after duplicates. Ranking with gaps
  resul:-
  1000 -1
  1000 -1
  200  -3
  300  -4
  DENSE_NUMBER() OVER (ORDER BY Salary DESC) --> No gap.Ranking without gaps
   resul:-
  1000 -1
  1000 -1
  200  -2
  300  -3

 16) Find salary difference from previous employee

 sql server buildin function : LAG() used to access previous row

  select empName,salary, salary -  LAG(salary) OVER (ORDER BY id) ) as PreviousSalary from employee 

 17) Find salary difference from next employee with Each department

  sql server buildin function : LEAD() used to access next row and PARTITION BY group department
  
  select empName, salary, salary - LEAD(salary) OVER (PARTITION BY department ORDER BY salary) as NextSalary from employee
